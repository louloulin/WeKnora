10.3 RegionResolver

// internal/application/service/region/resolver.go

type RegionResolver struct {
    bindings   map[uint64]*types.TenantRegionBinding  // tenant_id -> binding
    geoIP      *geoiplookup.CountryResolver
    mu         sync.RWMutex
}

// Resolve returns the target region for a request.
// Priority:
//  1. Tenant explicit binding (TenantRegionBinding.PrimaryRegion)
//  2. User IP GeoIP → nearest region
//  3. Default region (config)
func (r *RegionResolver) Resolve(ctx context.Context, tenantID uint64, ip string) (Region, error) {
    r.mu.RLock()
    defer r.mu.RUnlock()
    if binding, ok := r.bindings[tenantID]; ok {
        return binding.PrimaryRegion, nil
    }
    country, err := r.geoIP.Lookup(ip)
    if err == nil {
        return r.regionForCountry(country), nil
    }
    return types.RegionDev, nil  // fallback
}

// ComplianceGate checks whether a cross-region action is allowed by the tenant's policy.
func (r *RegionResolver) ComplianceGate(ctx context.Context, tenantID uint64, src, dst Region) error {
    binding := r.bindings[tenantID]
    if binding == nil {
        return nil  // no policy = allow
    }
    if !binding.ResidencyPolicy.AllowsCrossRegion(src, dst) {
        return ErrResidencyViolation
    }
    return nil
}

10.4 Middleware

// internal/middleware/region.go

// RegionEnforcer injects the resolved region into the request context
// and blocks requests whose target region doesn't match the tenant's
// residency policy.
func RegionEnforcer(resolver *region.Resolver) gin.HandlerFunc {
    return func(c *gin.Context) {
        tenantID := uint64FromCtx(c)
        ip := c.ClientIP()
        target, err := resolver.Resolve(c, tenantID, ip)
        if err != nil {
            c.AbortWithStatusJSON(500, gin.H{"error": "region_resolve_failed"})
            return
        }
        c.Set("region", target)

        // Check if the request is targeting a different region
        if requested := c.GetHeader("X-Target-Region"); requested != "" {
            if err := resolver.ComplianceGate(c, tenantID, target, types.Region(requested)); err != nil {
                // Audit log the violation
                audit.LogCrossRegion(c, target, types.Region(requested), false, err.Error())
                c.AbortWithStatusJSON(451, gin.H{"error": "residency_violation"})
                return
            }
        }
        c.Next()
    }
}

type PluginManifest struct {
    ID            string    `json:"id"`
    Version       string    `json:"version"`
    Author        string    `json:"author"`
    Signature     string    `json:"signature"`  // RSA-PSS signed manifest hash
    PublicKey     string    `json:"public_key"` // Plugin author's public key
    Permissions   []string  `json:"permissions"`
    MinWeKnoraVersion string `json:"min_weknora_version"`
    EntryPoint    string    `json:"entry_point"`  // e.g., "dist/index.js"
}

// VerifySignature uses the embedded public key to verify the manifest.
// Plugins without valid signatures are rejected at install time.
func (m *PluginManifest) VerifySignature() error {
    hash := sha256.Sum256(m.CanonicalBytes())
    return rsa.VerifyPSS(
        m.publicKey(),
        crypto.SHA256, hash[:],
        m.Signature, &rsa.PSSOptions{SaltLength: rsa.PSSSaltLengthAuto},
    )
}
