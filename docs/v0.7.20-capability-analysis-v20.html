关键结论

  1. 飞书文档是 WeKnora 最接近的对标对象：同样定位"知识管理 + AI 助手 + 协作"的一体化平台；WeKnora 在 RAG 与企业级方面领先，差距在 多维表格（Bitable） 与 妙搭（低代码 Agents）。
  2. v0.7.20 Synced Blocks 已落地（commit 287e1960），后端 100%、前端 40%；Tiptap 渲染待 v0.7.20.x。
  3. v0.7.21 是 P1 优先级：Custom Agents / Workers 是飞书妙搭、Notion Custom Agents、Glean Agents、Coda Packs 的通用对标，预计 30 天 / +3,000 行。
  4. v0.7.22 解决企业安全合规：DLP（数据丢失防护）+ AuthZ Admin UI 是金融/医疗/政企的硬性要求。
  5. v0.7.23 完善多维表：飞书多维表格（Bitable）是飞书增长最快的功能；WeKnora 现有 WikiDatabaseView 只有 5 视图，需要升级到 formula / rollup / linked record / automation。
  6. v0.7.24-v25 完善生态与基础设施：JS/Python/Go SDK + 插件签名 + 多地域容灾。

8.2 数据模型（迁移 114）

-- 114.1: custom_agents（agent 主表）
CREATE TABLE custom_agents (
    id              BIGSERIAL    PRIMARY KEY,
    tenant_id       BIGINT       NOT NULL,
    kb_id           VARCHAR(36)  NOT NULL,
    owner_id        BIGINT       NOT NULL,
    name            VARCHAR(128) NOT NULL,
    description     TEXT         NOT NULL DEFAULT '',
    system_prompt   TEXT         NOT NULL,
    -- 触发器类型: manual | cron | event | webhook
    trigger_type    VARCHAR(32)  NOT NULL DEFAULT 'manual',
    -- 触发器配置: JSON（cron expr / event name / webhook path）
    trigger_config  JSONB        NOT NULL DEFAULT '{}',
    -- 状态: draft | active | paused | archived
    status          VARCHAR(32)  NOT NULL DEFAULT 'draft',
    version         BIGINT       NOT NULL DEFAULT 1,
    timeout_seconds INT          NOT NULL DEFAULT 300,
    max_steps       INT          NOT NULL DEFAULT 20,
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    updated_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    UNIQUE (tenant_id, name)
);

-- 114.2: agent_tool_bindings（agent ↔ tool 关联）
CREATE TABLE agent_tool_bindings (
    id              BIGSERIAL    PRIMARY KEY,
    agent_id        BIGINT       NOT NULL,
    tenant_id       BIGINT       NOT NULL,
    tool_name       VARCHAR(128) NOT NULL,
    -- 凭据引用（指向 agent_credentials）
    credential_ref  VARCHAR(128),
    -- 工具配置（参数覆盖）
    config          JSONB        NOT NULL DEFAULT '{}',
    enabled         BOOLEAN      NOT NULL DEFAULT TRUE,
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    UNIQUE (agent_id, tool_name)
);

-- 114.3: agent_credentials（凭据 Vault）
CREATE TABLE agent_credentials (
    id              BIGSERIAL    PRIMARY KEY,
    tenant_id       BIGINT       NOT NULL,
    name            VARCHAR(128) NOT NULL,
    -- 凭据类型: api_key | oauth2 | basic | bearer
    credential_type VARCHAR(32)  NOT NULL,
    -- 加密后的密文（KMS / AES-256-GCM）
    ciphertext      BYTEA        NOT NULL,
    -- 加密元数据：算法、IV、auth tag
    enc_meta        JSONB        NOT NULL DEFAULT '{}',
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    UNIQUE (tenant_id, name)
);

-- 114.4: agent_runs（执行历史）
CREATE TABLE agent_runs (
    id              BIGSERIAL    PRIMARY KEY,
    tenant_id       BIGINT       NOT NULL,
    agent_id        BIGINT       NOT NULL,
    -- 触发类型: manual | cron | event | webhook | api
    triggered_by    VARCHAR(32)  NOT NULL,
    triggered_user  BIGINT,                -- nullable for system triggers
    -- 状态: queued | running | succeeded | failed | timeout | cancelled
    status          VARCHAR(32)  NOT NULL DEFAULT 'queued',
    input_payload      JSONB        NOT NULL DEFAULT '{}',
    output_payload     JSONB        NOT NULL DEFAULT '{}',
    error_message   TEXT         NOT NULL DEFAULT '',
    steps_count     INT          NOT NULL DEFAULT 0,
    -- 配额消耗（token / cost）
    tokens_used     BIGINT       NOT NULL DEFAULT 0,
    cost_micros     BIGINT       NOT NULL DEFAULT 0,  -- micro-USD (1 USD = 1,000,000 micros)
    started_at      TIMESTAMPTZ,
    finished_at     TIMESTAMPTZ,
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_agent_runs_tenant ON agent_runs (tenant_id, created_at DESC);
CREATE INDEX idx_agent_runs_agent  ON agent_runs (agent_id, created_at DESC);

-- 114.5: agent_credit_ledger（配额流水）
CREATE TABLE agent_credit_ledger (
    id              BIGSERIAL    PRIMARY KEY,
    tenant_id       BIGINT       NOT NULL,
    agent_id        BIGINT       NOT NULL,
    run_id          BIGINT,                          -- nullable for top-ups
    -- 操作类型: charge | refund | grant | expire
    operation       VARCHAR(32)  NOT NULL,
    -- 配额单位: tokens | invocations | cost_micros
    unit            VARCHAR(32)  NOT NULL,
    quantity        BIGINT       NOT NULL,           -- positive charge, negative refund
    balance_after    BIGINT       NOT NULL,
    -- 关联配额策略（versioned）
    policy_version  BIGINT       NOT NULL,
    notes           TEXT         NOT NULL DEFAULT '',
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_credit_ledger_tenant ON agent_credit_ledger (tenant_id, created_at DESC);
CREATE INDEX idx_credit_ledger_agent  ON agent_credit_ledger (agent_id, created_at DESC);

-- 114.6: agent_quota_policies（配额策略 + 版本化）

    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
