8.1 数据模型

-- 迁移 117 (pg + sqlite 032) — Doc ↔ KB 桥接
CREATE TABLE doc_kb_bindings (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL,
    knowledge_id BIGINT NOT NULL REFERENCES knowledges(id) ON DELETE CASCADE,
    chunk_id BIGINT NOT NULL REFERENCES chunks(id) ON DELETE CASCADE,
    wiki_page_id BIGINT REFERENCES wiki_pages(id) ON DELETE SET NULL,
    summary TEXT NOT NULL,
    keyphrases TEXT[],           -- 关键词数组
    auto_tags TEXT[],            -- 自动标签
    confidence REAL NOT NULL DEFAULT 0.0,
    bound_by BIGINT NOT NULL,
    bound_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_doc_kb_tenant ON doc_kb_bindings(tenant_id);
CREATE INDEX idx_doc_kb_chunk ON doc_kb_bindings(chunk_id);
CREATE INDEX idx_doc_kb_wiki ON doc_kb_bindings(wiki_page_id);

-- 迁移 118 (pg + sqlite 033) — Database / 多维表
CREATE TABLE wk_databases (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    schema JSONB NOT NULL,        -- 字段定义
    created_by BIGINT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE TABLE wk_database_rows (
    id BIGSERIAL PRIMARY KEY,
    database_id BIGINT NOT NULL REFERENCES wk_databases(id) ON DELETE CASCADE,
    tenant_id BIGINT NOT NULL,
    values JSONB NOT NULL,        -- 行数据
    created_by BIGINT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_wkdb_rows_db ON wk_database_rows(database_id);

-- 迁移 119 (pg + sqlite 034) — AI Assistant 对话
CREATE TABLE ai_assistant_threads (
    id BIGSERIAL PRIMARY KEY,
    tenant_id BIGINT NOT NULL,
    user_id BIGINT NOT NULL,
    title VARCHAR(255),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    last_active_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE TABLE ai_assistant_messages (
    id BIGSERIAL PRIMARY KEY,
    thread_id BIGINT NOT NULL REFERENCES ai_assistant_threads(id) ON DELETE CASCADE,
    role VARCHAR(32) NOT NULL,    -- user / assistant / tool
    content TEXT NOT NULL,
    tool_calls JSONB,
    citations JSONB,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
