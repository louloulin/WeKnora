二、数据模型（迁移 1）

-- 114.1: agent_triggers (cron / event / webhook / manual)
CREATE TABLE agent_triggers (
    id BIGSERIAL PRIMARY KEY, tenant_id BIGINT NOT NULL,
    agent_id VARCHAR(36) NOT NULL,
    trigger_type VARCHAR(32) NOT NULL, name VARCHAR(128) NOT NULL,
    trigger_config TEXT NOT NULL, payload_template TEXT NOT NULL,
    status VARCHAR(32) NOT NULL DEFAULT 'active',
    last_fired_at TIMESTAMPTZ, next_fire_at TIMESTAMPTZ,
    created_by BIGINT NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (tenant_id, agent_id, name)
);

-- 114.2: agent_runs (执行历史)
CREATE TABLE agent_runs (
    id BIGSERIAL PRIMARY KEY, tenant_id BIGINT NOT NULL,
    agent_id VARCHAR(36) NOT NULL, trigger_id BIGINT,
    triggered_by VARCHAR(32) NOT NULL, status VARCHAR(32) NOT NULL,
    input_payload TEXT, output_payload TEXT, error_message TEXT,
    steps_count INT, tokens_used BIGINT, cost_micros BIGINT,
    started_at TIMESTAMPTZ, finished_at TIMESTAMPTZ, duration_ms INT
);

-- 114.3: agent_credentials (AES-256-GCM Vault)
CREATE TABLE agent_credentials (
    id BIGSERIAL PRIMARY KEY, tenant_id BIGINT NOT NULL,
    name VARCHAR(128) NOT NULL, credential_type VARCHAR(32) NOT NULL,
    ciphertext BYTEA, nonce BYTEA, auth_tag BYTEA,
    expires_at TIMESTAMPTZ, last_used_at TIMESTAMPTZ
);
